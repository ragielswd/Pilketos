        // Hybrid RPC Engine Bridge for Apps Script and Standalone External Hosting
        const callBackend = (actionName, payload = {}) => {
          return new Promise((resolve, reject) => {
            // Mode A: Embedded in Google Apps Script (native google.script.run)
            if (typeof google !== 'undefined' && google.script && google.script.run) {
              const runner = google.script.run
                .withSuccessHandler((res) => resolve(res))
                .withFailureHandler((err) => reject(err));
              
              if (actionName === 'getInitialData') runner.getInitialData();
              else if (actionName === 'getQuickCountData') runner.getQuickCountData();
              else if (actionName === 'submitVote') runner.submitVote(payload.token, payload.candidateId);
              else if (actionName === 'saveCandidate') runner.saveCandidate(payload.candidate);
              else if (actionName === 'deleteCandidate') runner.deleteCandidate(payload.id);
              else if (actionName === 'importTokens') runner.importTokens(payload.tokens);
              else if (actionName === 'resetVotingData') runner.resetVotingData();
              else reject(new Error("Fungsi backend tidak ditemukan: " + actionName));
            } else {
              // Mode B: Standalone HTML / Deployed External Web App URL
              const apiUrl = window.GAS_API_URL || localStorage.getItem('gas_api_url');
              if (!apiUrl) {
                // Return offline fallback if API URL is not set
                console.warn("Google Apps Script URL belum diatur.");
                return resolve(null);
              }


              fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                body: JSON.stringify({ action: actionName, payload: payload })
              })
                .then(res => res.json())
                .then(data => resolve(data))
                .catch(err => reject(err));
            }
          });

                                    if (tokenStatusFilter === 'USED' && !isUsed) return false;
                                    if (tokenSearch && !token.toLowerCase().includes(tokenSearch.toLowerCase())) return false;
                                    return true;
                                  }).length === 0 ? (
                                    <div className="p-8 text-center text-xs text-slate-400 font-medium">
                                      Tidak ada token yang sesuai dengan filter atau pencarian.
                                    </div>
                                  ) : (
                                    <div className="grid grid-cols-2 sm:grid-cols-4 md:grid-cols-6 gap-2 p-3">
                                      {validTokens
                                        .filter((token) => {
                                          const isUsed = usedTokens.includes(token);
                                          if (tokenStatusFilter === 'UNUSED' && isUsed) return false;
                                          if (tokenStatusFilter === 'USED' && !isUsed) return false;
                                          if (tokenSearch && !token.toLowerCase().includes(tokenSearch.toLowerCase())) return false;
                                          return true;
                                        })
                                        .slice(0, 80)
                                        .map((token, i) => {
                                          const isUsed = usedTokens.includes(token);
                                          return (
                                            <div
                                              key={i}
                                              className={`p-2 rounded-lg text-xs font-mono font-semibold text-center truncate border ${
                                                isUsed ? 'bg-rose-100 text-rose-700 border-rose-200 line-through' : 'bg-white text-emerald-800 border-emerald-200'
                                              }`}
                                            >
                                              {token}
                                            </div>
                                          );
                                        })}
                                    </div>
                                  )}
                                </div>
                              </div>
                            </div>
                          </div>
                        )}
                      </div>
                    )}
                  </>
                )}
              </main>


              {/* Footer */}

                <div className="max-w-7xl mx-auto px-4 space-y-2">
                  <p className="font-semibold text-slate-300">
                    Sistem E-Voting Pemilihan Ketua OSIS • SMP Negeri 36 Samarinda
                  </p>
                  <p className="text-slate-500">
                    Terintegrasi Real-Time dengan Database Google Spreadsheet & Google Apps Script.
                  </p>
                </div>
              </footer>


              {/* MODALS */}
              
              {/* Delete Modal */}
              {candidateToDelete && (
                <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-sm flex items-center justify-center p-4">
                  <div className="bg-white rounded-2xl max-w-md w-full p-6 text-center space-y-6 shadow-2xl border border-slate-200">

                      <Trash2 className="w-8 h-8" />
                    </div>
                    <div className="space-y-2">
                      <h3 className="text-xl font-extrabold text-slate-900">Hapus Calon Ketua OSIS</h3>
                      <p className="text-sm text-slate-600">
                        Apakah Anda yakin ingin menghapus kandidat <br />
                        <span className="font-bold text-slate-900 text-base">
                          Nomor Urut {candidateToDelete.nomorUrut} - {candidateToDelete.nama}

                        Batal
                      </button>

                        Hapus Kandidat
                      </button>
                    </div>
                  </div>
                </div>
              )}


              {/* Detail Vision Mission Modal */}
              {detailModalCandidate && (
                <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-sm flex items-center justify-center p-4">
                  <div className="bg-white rounded-2xl max-w-lg w-full overflow-hidden shadow-2xl border border-slate-200">
                    <div className="bg-slate-900 p-4 text-white flex items-center justify-between">
                      <span className="text-xs font-bold text-amber-400 uppercase tracking-widest">
                        Detail Calon Ketua OSIS Nomor Urut {detailModalCandidate.nomorUrut}

                      <button onClick={() => setDetailModalCandidate(null)} className="p-1 text-slate-400 hover:text-white rounded-lg">
                        <X className="w-5 h-5" />
                      </button>
                    </div>


                    <div className="p-6 space-y-5 max-h-[80vh] overflow-y-auto">
                      <div className="flex items-center space-x-4">
                        <img src={detailModalCandidate.foto} alt={detailModalCandidate.nama} className="w-20 h-20 rounded-xl object-cover border-2 border-slate-200" />

                      <CheckCircle2 className="w-10 h-10" />

                    <div className="space-y-2">
                      <h3 className="text-2xl font-black text-slate-900">Terima Kasih Telah Memilih!</h3>

                        <div>
                          <label className="block text-xs font-bold uppercase text-slate-600 mb-1">Nomor Urut</label>

                        <button type="button" onClick={() => setCandidateModalOpen(false)} className="px-4 py-2 bg-slate-100 text-slate-700 font-bold rounded-xl text-xs">
                          Batal
                        </button>

                  </div>

              {/* Import Loading Modal */}
              {isImportingTokens && (
                <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4">
                  <div className="bg-white rounded-3xl max-w-md w-full p-8 text-center space-y-6 shadow-2xl border border-amber-200/80">
                    <div className="relative w-24 h-24 mx-auto flex items-center justify-center">
                      <div className="absolute inset-0 rounded-full bg-gradient-to-tr from-amber-400 via-blue-500 to-indigo-600 animate-spin opacity-75 blur-[2px]"></div>
                      <div className="absolute inset-1 rounded-full bg-white flex items-center justify-center shadow-inner">
                        <RefreshCw className="w-10 h-10 text-amber-500 animate-spin" />
                      </div>
                    </div>


                    <div className="space-y-2">
                      <h3 className="text-xl font-black text-slate-900 tracking-tight">
                        Mengimpor Database Token
                      </h3>
                      <div className="inline-block px-4 py-2 bg-amber-50 rounded-xl border border-amber-200">
                        <p className="text-xs text-amber-900 font-bold">{importLoadingMessage}</p>
                      </div>
                    </div>
                  </div>
                </div>
              )}


              {/* Reset Data Confirmation Modal */}
              {showResetModal && (
                <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-sm flex items-center justify-center p-4">
                  <div className="bg-white rounded-2xl max-w-md w-full p-6 text-center space-y-6 shadow-2xl border border-rose-200">
                    <div className="w-14 h-14 bg-rose-100 text-rose-600 rounded-full mx-auto flex items-center justify-center border-2 border-rose-200">
                      <AlertTriangle className="w-7 h-7" />
                    </div>


                    {resetStep === 1 ? (
                      <div className="space-y-4">
                        <h3 className="text-xl font-bold text-slate-900">Konfirmasi Reset Data Pemilihan</h3>
                        <p className="text-xs text-slate-600 leading-relaxed">
                          Tindakan ini akan mengosongkan data di Sheet 'Votes' dan me-reset status seluruh token di Sheet 'Tokens' menjadi UNUSED.
                        </p>
                        <div className="flex space-x-2 pt-2">
                          <button onClick={() => setShowResetModal(false)} className="w-1/2 py-2.5 bg-slate-100 text-slate-700 font-bold rounded-xl text-xs">
                            Batal
                          </button>

                      </div>
                    ) : (
                      <div className="space-y-4">
                        <h3 className="text-xl font-bold text-rose-600">Peringatan Terakhir!</h3>

                          Ketik kata <code className="bg-rose-100 text-rose-800 font-bold px-1 py-0.5 rounded">RESET</code> di bawah ini untuk mengonfirmasi:
                        </p>
                        <input
                          type="text"
                          value={resetConfirmText}
                          onChange={(e) => {
                            setResetConfirmText(e.target.value);
                            setResetStatusMsg('');
                          }}
                          placeholder="Ketik RESET"
                          className="w-full px-4 py-2 bg-slate-50 border border-slate-300 rounded-xl text-center text-sm font-mono font-bold"
                        />


                        {resetStatusMsg && <p className="text-xs text-rose-600 font-bold">{resetStatusMsg}</p>}


                        <div className="flex space-x-2 pt-2">
                          <button onClick={() => setShowResetModal(false)} className="w-1/2 py-2.5 bg-slate-100 text-slate-700 font-bold rounded-xl text-xs">
                            Batal
                          </button>
                          <button onClick={handleExecuteReset} className="w-1/2 py-2.5 bg-rose-600 text-white font-bold rounded-xl text-xs hover:bg-rose-700">
                            Hapus Permanen
                          </button>
                        </div>
                      </div>
                    )}
                  </div>
                </div>
              )}
            </div>
          );
        }
